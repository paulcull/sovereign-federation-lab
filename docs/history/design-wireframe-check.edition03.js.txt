const assert = require('node:assert/strict');
const m = require('../dist/app.js');
const transfer={region:'UK',storage:true,processing:true,entitlement:true,expired:false,fields:'minimal'};
assert.equal(m.evaluate(transfer).allow,true);
assert.equal(m.evaluate({...transfer,region:'US'}).allow,false);
for(const flag of ['storage','processing','entitlement'])assert.equal(m.evaluate({...transfer,[flag]:false}).allow,false);
assert.equal(m.evaluate({...transfer,expired:true}).allow,false);
assert(!m.evaluate({...transfer,fields:'restricted'}).fields.includes('nationalId'));
for(const stage of [1,2,4])assert.equal(m.nextRevocation(stage,false),stage);
assert.equal(m.nextRevocation(0,false),1);
assert.equal(m.nextRevocation(1,true),2);
assert.equal(m.nextRevocation(5,true),5);
const read={mode:'strict',elapsed:29,localDeny:false,stale:false};
assert.equal(m.outageRead(read).allow,true);
assert.equal(m.outageRead({...read,elapsed:30}).allow,false);
assert.equal(m.outageRead({...read,mode:'offline',elapsed:45}).allow,true);
assert.equal(m.outageRead({...read,mode:'offline',elapsed:120}).allow,false);
assert.equal(m.outageRead({...read,mode:'offline',localDeny:true}).allow,false);
assert.equal(m.outageRead({...read,mode:'offline',stale:true}).allow,false);
assert.equal(m.outageRead({...read,mode:'unknown'}).allow,false);
assert.equal(m.outageRead({...read,elapsed:NaN}).allow,false);
console.log('Passed: exclusion, field denial, expiry, revocation receipts and outage-read boundary checks.');
